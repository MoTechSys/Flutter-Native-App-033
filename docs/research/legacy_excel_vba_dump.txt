Private Sub UserForm_Initialize()
    ' تهيئة الشهور
    cboMonth.List = Array("يناير", "فبراير", "مارس", "أبريل", "مايو", "يونيو", _
                          "يوليو", "أغسطس", "سبتمبر", "أكتوبر", "نوفمبر", "ديسمبر")
    cboMonth.ListIndex = Month(Date) - 1  ' الشهر الحالي
    
    ' تهيئة السنوات (من 2020 إلى 2030)
    For Y = 2020 To 2030
        cboYear.AddItem Y
    Next Y
    cboYear.Text = Year(Date)  ' السنة الحالية
    
    ' تحديث التقويم
    Call UpdateCalendar
End Sub

Private Sub UpdateCalendar()
    lstDays.Clear
    
    Dim selectedMonth As Integer
    Dim selectedYear As Integer
    selectedMonth = cboMonth.ListIndex + 1
    selectedYear = Val(cboYear.Text)
    
    ' تحديد عدد أيام الشهر
    Dim daysInMonth As Integer
    daysInMonth = Day(DateSerial(selectedYear, selectedMonth + 1, 0))
    
    ' تعبئة ListBox بالأيام
    For i = 1 To daysInMonth
        lstDays.AddItem i
    Next i
End Sub

Private Sub cboMonth_Change()
    Call UpdateCalendar
End Sub

Private Sub cboYear_Change()
    Call UpdateCalendar
End Sub

Private Sub cmdOK_Click()
    If lstDays.ListIndex = -1 Then

    End If
    
    Dim selectedDay As Integer

    Dim selectedMonth As Integer
    selectedMonth = cboMonth.ListIndex + 1
    
    Dim selectedYear As Integer
    selectedYear = Val(cboYear.Text)
    
    selectedDate = DateSerial(selectedYear, selectedMonth, selectedDay)
    
    ' إخفاء النموذج
    Me.Hide
End Sub

Private Sub cmdSearch_Click()
    ' التحقق من إدخال التاريخ
    If txtDate.Text = "" Then
        MsgBox "الرجاء اختيار التاريخ أولاً", vbExclamation
        Exit Sub
    End If
    
    Dim selectedDate As Date
    On Error Resume Next ' إضافة معالجة الأخطاء
    selectedDate = CDate(txtDate.Text)
    If Err.Number <> 0 Then
        MsgBox "تنسيق التاريخ غير صحيح", vbExclamation
        Exit Sub
    End If
    On Error GoTo 0
    
    ' تفريغ الحقول قبل البحث
    txtQuantity.Text = ""
    txtDocumentNumber.Text = ""
    txtDetails.Text = ""
    
    ' البحث في ورقة "المولدات"
    Dim wsGenerators As Worksheet
    Set wsGenerators = ThisWorkbook.Sheets("المولدات")
    
    Dim foundInGenerators As Boolean
    foundInGenerators = False
    Dim generatorRow As Long
    
    Dim lastRowGenerators As Long
    lastRowGenerators = wsGenerators.Cells(wsGenerators.Rows.Count, "C").End(xlUp).Row
    
    ' البحث عن التاريخ في المولدات - التصحيح هنا
    Dim i As Long
    For i = 2 To lastRowGenerators
        If IsDate(wsGenerators.Cells(i, "C").Value) Then
            ' استخدم DateValue لمقارنة التواريخ فقط بدون الوقت
            If DateValue(wsGenerators.Cells(i, "C").Value) = DateValue(selectedDate) Then
                foundInGenerators = True
                generatorRow = i
                
                ' عرض الكمية إذا كانت موجودة
                If wsGenerators.Cells(i, "S").Value <> "" Then
                    txtQuantity.Text = wsGenerators.Cells(i, "S").Value
                End If
                Exit For
            End If
        End If
    Next i
    
    ' البحث في ورقة "المنظومه" - التصحيح هنا
    Dim wsSystem As Worksheet
    Set wsSystem = ThisWorkbook.Sheets("المنظومه")
    
    Dim foundInSystem As Boolean
    foundInSystem = False
    Dim systemRow As Long
    
    Dim lastRowSystem As Long
    lastRowSystem = wsSystem.Cells(wsSystem.Rows.Count, "B").End(xlUp).Row
    
    ' البحث عن التاريخ في المنظومه - التصحيح هنا
    Dim j As Long
    For j = 2 To lastRowSystem
        If IsDate(wsSystem.Cells(j, "B").Value) Then
            ' استخدم DateValue لمقارنة التواريخ فقط بدون الوقت
            If DateValue(wsSystem.Cells(j, "B").Value) = DateValue(selectedDate) Then
                foundInSystem = True
                systemRow = j
                
                ' عرض رقم السند إذا كان موجوداً - التصحيح هنا
                txtDocumentNumber.Text = wsSystem.Cells(j, "J").Value & ""
                
                ' عرض التفاصيل إذا كانت موجودة - التصحيح هنا
                txtDetails.Text = wsSystem.Cells(j, "K").Value & ""
                Exit For
            End If
        End If
    Next j
    
    ' إدارة أزرار الحفظ والتعديل
    Dim hasData As Boolean
    hasData = (txtQuantity.Text <> "" Or txtDocumentNumber.Text <> "" Or txtDetails.Text <> "")
    
    ' إذا وجدنا التاريخ ولكن بدون بيانات
    If (foundInGenerators Or foundInSystem) And Not hasData Then
        SaveData.Visible = True
        cmdEdit.Visible = False
        MsgBox "الرجاء ادخال الكميه الجديده ", vbInformation
    ' إذا وجدنا التاريخ وبها بيانات
    ElseIf hasData Then
        SaveData.Visible = False
        cmdEdit.Visible = True
        MsgBox "تم العثور على كميه بهاذا التاريخ وعرضها", vbInformation
    ' إذا لم نجد التاريخ
    Else
        SaveData.Visible = True
        cmdEdit.Visible = False
        MsgBox "هذا التاريخ ليس في سنة 2025 ", vbExclamation
    End If
End Sub

Private Sub SaveData_Click()
    ' التحقق من البيانات المدخلة
    If txtDate.Text = "" Then
        MsgBox "الرجاء اختيار التاريخ", vbExclamation
        Exit Sub
    End If
    
    If Not IsNumeric(txtQuantity.Text) Then
        MsgBox "الرجاء إدخال كمية صحيحة", vbExclamation
        Exit Sub
    End If
    
    If txtDocumentNumber.Text = "" Then
        MsgBox "الرجاء إدخال رقم السند", vbExclamation
        Exit Sub
    End If
    
    If txtDetails.Text = "" Then
        MsgBox "الرجاء إدخال تفاصيل الخطة", vbExclamation
        Exit Sub
    End If
    
    Dim selectedDate As Date
    selectedDate = CDate(txtDate.Text)
    
    ' العمل على ورقة "المولدات"
    Dim wsGenerators As Worksheet
    Set wsGenerators = ThisWorkbook.Sheets("المولدات")
    
    Dim foundInGenerators As Boolean
    foundInGenerators = False
    
    Dim lastRowGenerators As Long
    lastRowGenerators = wsGenerators.Cells(wsGenerators.Rows.Count, "C").End(xlUp).Row
    
    ' البحث في ورقة المولدات وإضافة الكمية
    For i = 2 To lastRowGenerators
        If IsDate(wsGenerators.Cells(i, "C").Value) Then
            If CDate(wsGenerators.Cells(i, "C").Value) = selectedDate Then
                foundInGenerators = True
                ' إضافة الكمية في العمود S
                wsGenerators.Cells(i, "S").Value = wsGenerators.Cells(i, "S").Value + Val(txtQuantity.Text)
                Exit For
            End If
        End If
    Next i
    
    ' العمل على ورقة "المنظومه"
    Dim wsSystem As Worksheet
    Set wsSystem = ThisWorkbook.Sheets("المنظومه")
    
    Dim lastRowSystem As Long
    lastRowSystem = wsSystem.Cells(wsSystem.Rows.Count, "A").End(xlUp).Row
    
    Dim foundInSystem As Boolean
    foundInSystem = False
    
    ' البحث في ورقة المنظومة عن التاريخ
    For j = 2 To lastRowSystem
        If IsDate(wsSystem.Cells(j, "B").Value) Then
            If CDate(wsSystem.Cells(j, "B").Value) = selectedDate Then
                foundInSystem = True
                ' إذا وجدنا التاريخ، نضع رقم السند في العمود J وتفاصيل الخطة في العمود K
                wsSystem.Cells(j, "J").Value = txtDocumentNumber.Text
                wsSystem.Cells(j, "K").Value = txtDetails.Text
                Exit For
            End If
        End If
    Next j
    
    ' إذا لم نجد التاريخ في ورقة المنظومة، نضيف سطرًا جديدًا
    If Not foundInSystem Then
        ' نجد الصف الفارغ التالي
        Dim newRow As Long
        newRow = lastRowSystem + 1
        
        ' نملأ البيانات
        wsSystem.Cells(newRow, "A").Value = selectedDate ' التاريخ في العمود A
        wsSystem.Cells(newRow, "J").Value = txtDocumentNumber.Text ' رقم السند في العمود J
        wsSystem.Cells(newRow, "K").Value = txtDetails.Text ' التفاصيل في العمود K
    End If
    
    MsgBox "تم حفظ البيانات بنجاح", vbInformation
    
    ' تفريغ الحقول بعد الحفظ
    txtDate.Text = ""
    txtQuantity.Text = ""
    txtDocumentNumber.Text = ""
    txtDetails.Text = ""
    
    ' إعادة تعيين أزرار الحفظ والتعديل
    SaveData.Visible = True
    cmdEdit.Visible = False
End Sub

Private Sub btnSearch_Click()

    Dim searchDate As String
    Dim ws As Worksheet
    Dim LastRow As Long
    Dim i As Long
    
    Set ws = SheetCAT ' استخدم اسم الورقة الفعلي
    searchDate = txtDate.Value
    
    If searchDate = "" Then
        MsgBox "الرجاء اختيار تاريخ للبحث", vbExclamation
        Exit Sub
    End If
    
    LastRow = ws.Cells(ws.Rows.Count, "C").End(xlUp).Row
    currentRow = 0
    
    ' البحث في العمود A عن التاريخ المحدد

Private Sub save_Click()
    ' التحقق من وجود أي بيانات في الحقول
    Dim hasData As Boolean
    hasData = False
    
    ' التحقق من وجود بيانات في مجموعة C18
    If diesel18.Value <> "" Or readNew18.Value <> "" Or _
       readNewScreen18.Value <> "" Or hourNow18.Value <> "" Then
        hasData = True
    End If
    
    ' التحقق من وجود بيانات في مجموعة C15
    If diesel15.Value <> "" Or readNew15.Value <> "" Or _
       readNewScreen15.Value <> "" Or hourNow15.Value <> "" Then
        hasData = True
    End If
    
    ' إذا لم يكن هناك أي بيانات
    If Not hasData Then
        MsgBox "لا توجد بيانات للحفظ. الرجاء إدخال البيانات أولاً.", vbExclamation
        Exit Sub
    End If
    
    ' التحقق من اكتمال البيانات
    Dim group18Complete As Boolean
    Dim group15Complete As Boolean
    Dim ws As Worksheet
    
    Set ws = SheetCAT
    
    ' التحقق من اكتمال مجموعة C18
    group18Complete = (diesel18.Value <> "" And readNew18.Value <> "" And _
                       readNewScreen18.Value <> "" And hourNow18.Value <> "")
    
    ' التحقق من اكتمال مجموعة C15
    group15Complete = (diesel15.Value <> "" And readNew15.Value <> "" And _
                       readNewScreen15.Value <> "" And hourNow15.Value <> "")
    
    ' التعديل على البيانات الموجودة
    If currentRow > 0 Then
        If group18Complete Then
            ws.Cells(currentRow, "D").Value = diesel18.Value
            ws.Cells(currentRow, "G").Value = readNew18.Value
            ws.Cells(currentRow, "M").Value = readNewScreen18.Value
            ws.Cells(currentRow, "U").Value = hourNow18.Value
        End If
        
        If group15Complete Then
            If currentRow + 1 > ws.Cells(ws.Rows.Count, "C").End(xlUp).Row Then
                ws.Rows(currentRow + 1).Insert
                ws.Cells(currentRow + 1, "C").Value = txtDate.Value
            End If
            ws.Cells(currentRow + 1, "D").Value = diesel15.Value
            ws.Cells(currentRow + 1, "G").Value = readNew15.Value
            ws.Cells(currentRow + 1, "M").Value = readNewScreen15.Value
            ws.Cells(currentRow + 1, "U").Value = hourNow15.Value
            ws.Cells(currentRow + 1, "C").Value = txtDate.Value
        End If
        
        MsgBox "تم تعديل البيانات بنجاح", vbInformation
        ClearFormFields
        UserForm_Initialize
        ThisWorkbook.save
        currentRow = 0
        Exit Sub
    End If
    
    ' إضافة بيانات جديدة
    Dim LastRow As Long
    Dim newRow As Long
    
    LastRow = ws.Cells(ws.Rows.Count, "D").End(xlUp).Row
    
    If ws.Cells(LastRow, "D").Value = "" Then
        newRow = LastRow
    Else
        newRow = LastRow + 1

    Dim pdfPath As String
    Dim originalPrinter As String
    Dim pdfPrinter As String
    Dim printerSet As Boolean
    Dim rngToPrint As Range
    
    ' حفظ إعدادات الطابعة الأصلية
    originalPrinter = Application.ActivePrinter
    printerSet = False
    
    ' محاولة تعيين طابعة PDF
    pdfPrinter = "AnyDesk Printer"
    On Error Resume Next
    Application.ActivePrinter = pdfPrinter
    If Err.Number = 0 Then printerSet = True
    On Error GoTo ErrorHandler
    
    ' إنشاء مسار الحفظ على سطح المكتب
    pdfPath = Environ("USERPROFILE") & "\Desktop\Backup\"
    If Dir(pdfPath, vbDirectory) = "" Then MkDir pdfPath
    pdfPath = pdfPath & "تقرير فني المولدات غليل " & "_" & Format(Now, "yyyy-mm-dd") & ".pdf"
    
    ' حفظ الإعدادات الأصلية للأوراق
    Set originalSettings = CreateObject("Scripting.Dictionary")
    
    For Each ws In ThisWorkbook.Worksheets
        ' تخطي الأوراق المخفية
        If ws.Visible <> xlSheetVisible Then
            GoTo SkipSheet
        End If
        
        ' حفظ الإعدادات الحالية
        originalSettings.Add ws.Name, Array( _
            ws.PageSetup.printArea, _
            ws.PageSetup.Orientation, _
            ws.PageSetup.PaperSize, _
            ws.PageSetup.Zoom, _
            ws.PageSetup.FitToPagesWide, _
            ws.PageSetup.FitToPagesTall _
        )
        
        ' تحديد منطقة الطباعة بشكل ديناميكي
        With ws
            ' تحديد الصف الأخير من العمود D
          LastRow = .Columns("D").Find("*", , xlValues, , xlByRows, xlPrevious).Row
                If LastRow < 1 Then LastRow = 1
            
            ' تحديد العمود الأخير بشكل ديناميكي (لجميع الصفوف)
            lastCol = .Cells.Find("*", .Range("A1"), xlFormulas, , xlByColumns, xlPrevious).Column
            If lastCol < 1 Then lastCol = 1
            
            ' تحديد نطاق الطباعة
            Set rngToPrint = .Range("A1", .Cells(LastRow, lastCol))
        End With
        
        ' تطبيق الإعدادات الجديدة
        With ws.PageSetup
            .printArea = rngToPrint.Address
            .Orientation = xlLandscape
            .PaperSize = xlPaperA4
            .Zoom = False
            .FitToPagesWide = 1
            .FitToPagesTall = False
            
            ' إعدادات هوامش
            .LeftMargin = Application.InchesToPoints(0.25)
            .RightMargin = Application.InchesToPoints(0.25)
            .TopMargin = Application.InchesToPoints(0.75)
            .BottomMargin = Application.InchesToPoints(0.75)
        End With
        
        ' تأكيد تطبيق إعدادات الطباعة
        ws.PageSetup.printArea = rngToPrint.Address
        
SkipSheet:
    Next ws
    
    ' تصدير إلى PDF
    ThisWorkbook.ExportAsFixedFormat _
        Type:=xlTypePDF, _
        Filename:=pdfPath, _
        Quality:=xlQualityStandard, _
        IncludeDocProperties:=True, _
        IgnorePrintAreas:=False, _
        OpenAfterPublish:=False
    
    ' استعادة الإعدادات الأصلية للأوراق
    For Each ws In ThisWorkbook.Worksheets
        If originalSettings.Exists(ws.Name) Then
            Dim settings As Variant
            settings = originalSettings(ws.Name)
            
            With ws.PageSetup
                .printArea = settings(0)
                .Orientation = settings(1)
                .PaperSize = settings(2)
                .Zoom = settings(3)
                .FitToPagesWide = settings(4)
                .FitToPagesTall = settings(5)
            End With
        End If
    Next ws
    
    ' إظهار رسالة نجاح
    MsgBox "تم تصدير الملف بنجاح إلى:" & vbCrLf & pdfPath, vbInformation, "تصدير PDF"
    
Cleanup:
    ' استعادة الطابعة الأصلية
    If originalPrinter <> "" Then
        Application.ActivePrinter = originalPrinter
    End If
    Exit Sub
    
ErrorHandler:
    ' معالجة الأخطاء
    MsgBox "حدث خطأ في الورقة '" & ws.Name & "': " & Err.Description, vbCritical
    
    ' استعادة إعدادات الأوراق في حالة خطأ
    If Not originalSettings Is Nothing Then
        For Each ws In ThisWorkbook.Worksheets
            If originalSettings.Exists(ws.Name) Then
                settings = originalSettings(ws.Name)

                    .printArea = settings(0)

                    .FitToPagesTall = settings(5)
                End With
            End If
        Next ws
    End If
    
    ' استعادة الطابعة الأصلية
    If originalPrinter <> "" Then
        Application.ActivePrinter = originalPrinter
    End If
End Sub

Private Sub save_Click()
    ' التحقق من اكتمال بيانات المجموعات
    Dim group18Complete As Boolean
    Dim group15Complete As Boolean
    
    ' التحقق من اكتمال مجموعة C18
    group18Complete = (diesel18.Value <> "" And readNew18.Value <> "" And _
                       readNewScreen18.Value <> "" And hourNow18.Value <> "")
    
    ' التحقق من اكتمال مجموعة C15
    group15Complete = (diesel15.Value <> "" And readNew15.Value <> "" And _
                       readNewScreen15.Value <> "" And hourNow15.Value <> "")
    
    ' الخروج إذا لم تكن هناك بيانات كاملة
    If Not group18Complete And Not group15Complete Then
        MsgBox "لا توجد بيانات كاملة للحفظ. الرجاء إدخال جميع الحقول لكل مجموعة.", vbExclamation
        Exit Sub
    End If
    
    ' التحقق من المجموعات غير المكتملة
    Dim group18Partial As Boolean
    Dim group15Partial As Boolean
    
    group18Partial = (diesel18.Value <> "" Or readNew18.Value <> "" Or _
                      readNewScreen18.Value <> "" Or hourNow18.Value <> "")
    
    group15Partial = (diesel15.Value <> "" Or readNew15.Value <> "" Or _
                      readNewScreen15.Value <> "" Or hourNow15.Value <> "")
    
    ' إظهار تحذير إذا كانت هناك مجموعة غير مكتملة
    If group18Partial And Not group18Complete Then
        MsgBox "تحذير: مجموعة C18 غير مكتملة. لن يتم حفظها.", vbExclamation
    End If
    
    If group15Partial And Not group15Complete Then
        MsgBox "تحذير: مجموعة C15 غير مكتملة. لن يتم حفظها.", vbExclamation
    End If
    
    Dim LastRow As Long
    Dim newRow As Long
    
    ' البحث عن أول صف فارغ في العمود D
    LastRow = SheetCAT.Cells(Rows.Count, "D").End(xlUp).Row
    
    ' تحديد الصف الجديد للإضافة
    If SheetCAT.Cells(LastRow, "D").Value = "" Then
        newRow = LastRow ' إذا كان العمود فارغاً تماماً
    Else
        newRow = LastRow + 1 ' إذا وجدت بيانات
    End If
    
    ' إضافة بيانات C18 إذا كانت مكتملة
    If group18Complete Then
        SheetCAT.Cells(newRow, "D").Value = diesel18.Value
        SheetCAT.Cells(newRow, "G").Value = readNew18.Value
        SheetCAT.Cells(newRow, "M").Value = readNewScreen18.Value
        SheetCAT.Cells(newRow, "U").Value = hourNow18.Value
    End If
    
    ' إضافة بيانات C15 إذا كانت مكتملة
    If group15Complete Then
        ' إذا كانت هناك بيانات C18 مكتملة نضيف C15 في الصف التالي
        ' وإلا نضيفها في نفس الصف
        Dim targetRow As Long
        targetRow = IIf(group18Complete, newRow + 1, newRow)
        
        SheetCAT.Cells(targetRow, "D").Value = diesel15.Value
        SheetCAT.Cells(targetRow, "G").Value = readNew15.Value
        SheetCAT.Cells(targetRow, "M").Value = readNewScreen15.Value
        SheetCAT.Cells(targetRow, "U").Value = hourNow15.Value
    End If
    
    ' تنظيف حقول C18 إذا تم حفظها
    If group18Complete Then
        readNew18.Value = ""
        readNewScreen18.Value = ""
        hourNow18.Value = ""
        diesel18.Value = ""
    End If
    
    ' تنظيف حقول C15 إذا تم حفظها
    If group15Complete Then
        readNew15.Value = ""
        readNewScreen15.Value = ""
        hourNow15.Value = ""
        diesel15.Value = ""
    End If

    If group18Complete Or group15Complete Then
        MsgBox message, vbInformation
    End If

Private Sub cmdLogin_Click()
    Dim ws As Worksheet
    Dim userRow As Long
    Dim found As Boolean
    
    Set ws = ThisWorkbook.Sheets("المستخدمين")
    found = False
    
    ' البحث عن المستخدم في القائمة
    For userRow = 2 To ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
        If LCase(ws.Cells(userRow, 1).Value) = LCase(txtUsername.Value) Then
            If ws.Cells(userRow, 2).Value = txtPassword.Value Then
                found = True
                Exit For
            End If
        End If
    Next userRow
    
    If found Then
        MsgBox "تم الدخول بنجاح!", vbInformation
        ' الانتقال للواجهة الرئيسية
       LoginForm.Hide
        FormOne.Show
        Unload Me
        
    Else
        MsgBox "بيانات الدخول غير صحيحة", vbCritical
        txtPassword.Value = ""
    End If
End Sub
